local HAS_SETTINGS = false
pcall(function()
	local s = settings()
	HAS_SETTINGS = (type(s) == "table" or type(s) == "userdata")
end)
local function setR(k, v)
	if not HAS_SETTINGS then return end
	pcall(function() settings().Rendering[k] = v end)
end

--=====================================================================
-- TEXTURAS
--=====================================================================
local function aplicarTexturas(on)
	CFG.TexMin = on
	if not on then return end
	setR("QualityLevel", Enum.QualityLevel.Level01)
	setR("MeshPartDetail", Enum.MeshPartDetailLevel.Level01)
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") and not d:IsA("Terrain") then
			if BK.mat[d] == nil then BK.mat[d] = d.Material end
			if BK.refl[d] == nil then BK.refl[d] = d.Reflectance end
			try(function()
				d.Material = Enum.Material.SmoothPlastic
				d.MaterialVariant = ""
				d.Reflectance = 0
			end)
		end
	end
end

--=====================================================================
-- FPS
--=====================================================================
local fpsConn
local function aplicarFPS(n)
	CFG.FPS = n
	if fpsConn then fpsConn:Disconnect(); fpsConn = nil end
	try(function() UGS.FramerateCap = n end)
	local nativo = try(function()
		if type(setfpscap) == "function" then setfpscap(n); return true end
		return false
	end)
	if not nativo then
		local alvo = 1 / n
		local last = os.clock()
		fpsConn = bind(RunService.RenderStepped:Connect(function()
			local now = os.clock()
			local dif = now - last
			if dif < alvo then task.wait(alvo - dif) end
			last = os.clock()
		end))
	end
end

--=====================================================================
-- SOMBRAS
--=====================================================================
local function aplicarSombras(off)
	CFG.SombrasOff = off
	try(function() Lighting.GlobalShadows = not off end)
	try(function() Lighting.ShadowSoftness = 0 end)
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") then
			if BK.shadow[d] == nil then BK.shadow[d] = d.CastShadow end
			try(function() d.CastShadow = not off end)
		end
	end
end

--=====================================================================
-- ANIMAÇÕES
--=====================================================================
local animConn = setmetatable({}, {__mode = "k"})

local function pararTracks(obj)
	local ok, tracks = pcall(function() return obj:GetPlayingAnimationTracks() end)
	if ok and tracks then
		for _, t in ipairs(tracks) do pcall(function() t:Stop(0) end) end
	end
end

local function hookAnim(obj)
	if animConn[obj] then return end
	local ok, c = pcall(function()
		return obj.AnimationPlayed:Connect(function(track)
			if CFG.AnimOff then
				task.defer(function() pcall(function() track:Stop(0) end) end)
			end
		end)
	end)
	if ok then animConn[obj] = c end
	pararTracks(obj)
end

local function aplicarAnim(off)
	CFG.AnimOff = off
	if not off then return end
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("Animator") or d:IsA("AnimationController") then hookAnim(d) end
	end
	if LP.Character then
		for _, d in ipairs(LP.Character:GetDescendants()) do
			if d:IsA("Animator") then hookAnim(d) end
		end
	end
end

bind(LP.CharacterAdded:Connect(function(char)
	task.spawn(function()
		local hum = char:WaitForChild("Humanoid", 10)
		if not hum then return end
		local anim = hum:WaitForChild("Animator", 10)
		if anim and CFG.AnimOff then hookAnim(anim) end
	end)
end))

local function aplicarQualidade(nome)

--=====================================================================
-- RENDER + CULLING
--=====================================================================
local function aplicarRender(dist)
	CFG.Dist = dist
	try(function()
		if Workspace.StreamingEnabled then Workspace.StreamingTargetRadius = dist end
	end)
	if BK.light.fogS == nil then
		BK.light.fogS = Lighting.FogStart
		BK.light.fogE = Lighting.FogEnd
	end
	try(function()
		Lighting.FogStart = dist * 0.55
		Lighting.FogEnd   = dist * 1.05
	end)
end

local parts, charSet, idx = {}, {}, 1

local function rebuild()
	local t, n = {}, 0
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") and not d:IsA("Terrain") then
			n += 1
			t[n] = d
		end
	end
	parts, idx = t, 1
	charSet = {}
	if LP.Character then
		for _, d in ipairs(LP.Character:GetDescendants()) do charSet[d] = true end
	end
end

local function limparCulling()
	for _, p in ipairs(parts) do
		try(function() p.LocalTransparencyModifier = 0 end)
	end
end

bind(RunService.Heartbeat:Connect(function()
	if not CFG.Culling or #parts == 0 or not Cam then return end
	local cp   = Cam.CFrame.Position
	local lim2 = CFG.Dist * CFG.Dist
	local n    = #parts
	local step = math.max(150, math.ceil(n / 6))
	for _ = 1, step do
		idx += 1
		if idx > n then idx = 1 end
		local p = parts[idx]
		if p and p.Parent then
			local want = 0
			if not charSet[p] then
				local d = p.Position - cp
				if (d.X*d.X + d.Y*d.Y + d.Z*d.Z) > lim2 then want = 1 end
			end
			if p.LocalTransparencyModifier ~= want then
				p.LocalTransparencyModifier = want
			end
		end
	end
end))

task.spawn(function()
	while true do
		task.wait(4)
		if CFG.Culling then rebuild() end
	end
end)

-- BRUTAL
--=====================================================================
local function aplicarBrutal(on)
	CFG.Brutal = on
	if not on then return end
	for _, e in ipairs(Lighting:GetChildren()) do
		if e:IsA("PostEffect") or e:IsA("BloomEffect") or e:IsA("BlurEffect")
		or e:IsA("DepthOfFieldEffect") or e:IsA("SunRaysEffect")
		or e:IsA("ColorCorrectionEffect") then
			if BK.fx[e] == nil then BK.fx[e] = e.Enabled end
			try(function() e.Enabled = false end)
		elseif e:IsA("Atmosphere") then
			if BK.fx[e] == nil then BK.fx[e] = e.Density end
			try(function() e.Density = 0 end)
		end
	end
	try(function()
		Lighting.EnvironmentDiffuseScale  = 0
		Lighting.EnvironmentSpecularScale = 0
	end)
	local ter = Workspace:FindFirstChildOfClass("Terrain")
	if ter then
		BK.terrain.dec = ter.Decoration
		BK.terrain.ws  = ter.WaterWaveSize
		BK.terrain.wv  = ter.WaterWaveSpeed
		BK.terrain.wr  = ter.WaterReflectance
		try(function()
			ter.Decoration = false
			ter.WaterWaveSize = 0
			ter.WaterWaveSpeed = 0
			ter.WaterReflectance = 0
		end)
	end
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ParticleEmitter") then
			if d.Rate > 15 then d.Rate = 15 end
		elseif d:IsA("Trail") or d:IsA("Beam") then
			try(function() d.Enabled = false end)
		elseif d:IsA("Smoke") or d:IsA("Fire") or d:IsA("Sparkles") then
			try(function() d.Enabled = false end)
		end
	end
	try(function() StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.PlayerList, false) end)
end

bind(Workspace.DescendantAdded:Connect(function(d)
	if d:IsA("BasePart") then
		if CFG.SombrasOff then try(function() d.CastShadow = false end) end
		if CFG.TexMin then
			try(function()
				d.Material = Enum.Material.SmoothPlastic
				d.Reflectance = 0
			end)
		end
	elseif d:IsA("Animator") or d:IsA("AnimationController") then
		if CFG.AnimOff then hookAnim(d) end
	elseif d:IsA("ParticleEmitter") and CFG.Brutal then
		if d.Rate > 15 then d.Rate = 15 end
	elseif (d:IsA("Trail") or d:IsA("Beam")) and CFG.Brutal then
		try(function() d.Enabled = false end)
	end
end))

--=====================================================================
-- AÇÕES

local function aplicarTudo()
	aplicarQualidade(CFG.Qual)
	aplicarTexturas(CFG.TexMin)
	aplicarFPS(CFG.FPS)
	aplicarSombras(CFG.SombrasOff)
	aplicarAnim(CFG.AnimOff)
	aplicarRender(CFG.Dist)
	if CFG.Brutal then aplicarBrutal(true) end
	if CFG.Culling then rebuild() end
end

local function restaurar()
	if fpsConn then fpsConn:Disconnect(); fpsConn = nil end
	for e, v in pairs(BK.fx) do
		if typeof(e) == "Instance" and e.Parent then
			pcall(function() e.Enabled = v end)
			pcall(function() e.Density = v end)
		end
	end
	BK.fx = {}
	if BK.light.fogS then
		pcall(function() Lighting.FogStart = BK.light.fogS end)
		pcall(function() Lighting.FogEnd = BK.light.fogE end)
	end
	try(function() Lighting.GlobalShadows = true end)
	try(function() Lighting.EnvironmentDiffuseScale = 1 end)
	try(function() Lighting.EnvironmentSpecularScale = 1 end)
	for p, m in pairs(BK.mat) do if p.Parent then pcall(function() p.Material = m end) end end
	for p, r in pairs(BK.refl) do if p.Parent then pcall(function() p.Reflectance = r end) end end
	for p, s in pairs(BK.shadow) do if p.Parent then pcall(function() p.CastShadow = s end) end end
	BK.mat, BK.refl, BK.shadow = setmetatable({},{__mode="k"}), setmetatable({},{__mode="k"}), setmetatable({},{__mode="k"})
	for t in pairs(animConn) do pcall(function() animConn[t]:Disconnect() end) end
	animConn = setmetatable({}, {__mode = "k"})
	limparCulling()
end

	if gp then return end
	if i.KeyCode == Enum.KeyCode.RightShift then main.Visible = not main.Visible end
end))

-- contador de FPS
local frames, acc = 0, 0
bind(RunService.RenderStepped:Connect(function(dt)
	frames += 1
	acc += dt
	if acc >= 0.5 then
		lblFps.Text = "FPS: " .. math.floor(frames / acc + 0.5)
		frames, acc = 0, 0
	end
end))
