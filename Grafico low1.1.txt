local okRun, errRun = xpcall(function()

local CONNS = {}
local function bind(c) CONNS[#CONNS+1] = c return c end
local function try(f, ...) local o, r = pcall(f, ...) if o then return r end end

local CFG = {
	FPS = 60, Qual = "BAIXA", Dist = 350,
	SombrasOff = true, AnimOff = true, TexMin = true,
	Brutal = true, Culling = true, AutoFechar = true,
}

local QUAL = {
	["EXTREMA BAIXA"] = { lvl = Enum.QualityLevel.Level01, saved = Enum.SavedQualitySetting.QualityLevel1 },
	["BAIXA"]         = { lvl = Enum.QualityLevel.Level03, saved = Enum.SavedQualitySetting.QualityLevel3 },
	["NORMAL"]        = { lvl = Enum.QualityLevel.Level05, saved = Enum.SavedQualitySetting.QualityLevel5 },
}

local BK = { fx = {}, light = {},
	mat = setmetatable({}, {__mode="k"}),
	refl = setmetatable({}, {__mode="k"}),
	sh = setmetatable({}, {__mode="k"}) }

local HAS_S = false
pcall(function() HAS_S = settings() ~= nil end)
local function setR(k, v) if HAS_S then pcall(function() settings().Rendering[k] = v end) end end

local function aplicarTexturas(on)
	CFG.TexMin = on
	if not on then return end
	setR("QualityLevel", Enum.QualityLevel.Level01)
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") and not d:IsA("Terrain") then
			if BK.mat[d] == nil then BK.mat[d] = d.Material end
			if BK.refl[d] == nil then BK.refl[d] = d.Reflectance end
			try(function() d.Material = Enum.Material.SmoothPlastic d.Reflectance = 0 end)
		end
	end
end

local fpsConn
local function aplicarFPS(n)
	CFG.FPS = n
	if fpsConn then fpsConn:Disconnect() fpsConn = nil end
	try(function() UGS.FramerateCap = n end)
	local nat = try(function()
		if type(setfpscap) == "function" then setfpscap(n) return true end
		return false
	end)
	if not nat then
		local alvo, last = 1/n, os.clock()
		fpsConn = bind(RunService.RenderStepped:Connect(function()
			local now = os.clock() local dif = now - last
			if dif < alvo then task.wait(alvo - dif) end
			last = os.clock()
		end))
	end
end

local function aplicarSombras(off)
	CFG.SombrasOff = off
	try(function() Lighting.GlobalShadows = not off end)
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") then
			if BK.sh[d] == nil then BK.sh[d] = d.CastShadow end
			try(function() d.CastShadow = not off end)
		end
	end
end

local animConn = setmetatable({}, {__mode="k"})
local function pararTracks(o)
	local ok, t = pcall(function() return o:GetPlayingAnimationTracks() end)
	if ok and t then for _, x in ipairs(t) do pcall(function() x:Stop(0) end) end end
end
local function hookAnim(o)
	if animConn[o] then return end
	local ok, c = pcall(function()
		return o.AnimationPlayed:Connect(function(tr)
			task.defer(function() pcall(function() tr:Stop(0) end) end)
		end)
	end)
	if ok then animConn[o] = c end
	pararTracks(o)
end
local function aplicarAnim(off)
	CFG.AnimOff = off
	if not off then return end
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("Animator") or d:IsA("AnimationController") then hookAnim(d) end
	end
end

local function aplicarQualidade(nome)
	local q = QUAL[nome]
	if not q then return end
	CFG.Qual = nome
	setR("QualityLevel", q.lvl)
	try(function() UGS.SavedQualityLevel = q.saved end)
end

local function aplicarRender(dist)
	CFG.Dist = dist
	try(function() if Workspace.StreamingEnabled then Workspace.StreamingTargetRadius = dist end end)
	if BK.light.fs == nil then BK.light.fs = Lighting.FogStart BK.light.fe = Lighting.FogEnd end
	try(function() Lighting.FogStart = dist * 0.55 Lighting.FogEnd = dist * 1.05 end)
end

local parts, ch, idx = {}, {}, 1
local function rebuild()
	local t, n = {}, 0
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") and not d:IsA("Terrain") then n += 1 t[n] = d end
	end
	parts, idx, ch = t, 1, {}
	if LP.Character then
		for _, d in ipairs(LP.Character:GetDescendants()) do ch[d] = true end
	end
end
local function limparCulling()
	for _, p in ipairs(parts) do try(function() p.LocalTransparencyModifier = 0 end) end
end
bind(RunService.Heartbeat:Connect(function()
	if not CFG.Culling or #parts == 0 or not Workspace.CurrentCamera then return end
	local cp = Workspace.CurrentCamera.CFrame.Position
	local l2 = CFG.Dist * CFG.Dist
	local n = #parts
	for _ = 1, math.max(150, math.ceil(n/6)) do
		idx += 1 if idx > n then idx = 1 end
		local p = parts[idx]
		if p and p.Parent then
			local w = 0
			if not ch[p] then
				local v = p.Position - cp
				if v.X*v.X + v.Y*v.Y + v.Z*v.Z > l2 then w = 1 end
			end
			if p.LocalTransparencyModifier ~= w then p.LocalTransparencyModifier = w end
		end
	end
end))

local function aplicarBrutal(on)
	CFG.Brutal = on
	if not on then return end
	for _, e in ipairs(Lighting:GetChildren()) do
		if e:IsA("PostEffect") then
			if BK.fx[e] == nil then BK.fx[e] = e.Enabled end
			try(function() e.Enabled = false end)
		elseif e:IsA("Atmosphere") then
			if BK.fx[e] == nil then BK.fx[e] = e.Density end
			try(function() e.Density = 0 end)
		end
	end
	try(function() Lighting.EnvironmentDiffuseScale = 0 Lighting.EnvironmentSpecularScale = 0 end)
	local ter = Workspace:FindFirstChildOfClass("Terrain")
	if ter then try(function()
		ter.Decoration = false ter.WaterWaveSize = 0 ter.WaterWaveSpeed = 0 ter.WaterReflectance = 0
	end) end
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ParticleEmitter") then
			if d.Rate > 15 then d.Rate = 15 end
		elseif d:IsA("Trail") or d:IsA("Beam") or d:IsA("Smoke") or d:IsA("Fire") or d:IsA("Sparkles") then
			try(function() d.Enabled = false end)
		end
	end
	try(function() StarterGui:SetCoreGuiEnabled(Enum.CoreGuiType.PlayerList, false) end)
end

bind(Workspace.DescendantAdded:Connect(function(d)
	if d:IsA("BasePart") then
		if CFG.SombrasOff then try(function() d.CastShadow = false end) end
		if CFG.TexMin then try(function() d.Material = Enum.Material.SmoothPlastic d.Reflectance = 0 end) end
	elseif d:IsA("Animator") or d:IsA("AnimationController") then
		if CFG.AnimOff then hookAnim(d) end
	end
end))

local function restaurar()
	if fpsConn then fpsConn:Disconnect() fpsConn = nil end
	for e, v in pairs(BK.fx) do
		if typeof(e) == "Instance" and e.Parent then
			pcall(function() e.Enabled = v end)
		end
	end
	BK.fx = {}
	if BK.light.fs then
		pcall(function() Lighting.FogStart = BK.light.fs end)
		pcall(function() Lighting.FogEnd = BK.light.fe end)
	end
	try(function() Lighting.GlobalShadows = true Lighting.EnvironmentDiffuseScale = 1 Lighting.EnvironmentSpecularScale = 1 end)
	for p, m in pairs(BK.mat) do if p.Parent then pcall(function() p.Material = m end) end end
	for p, r in pairs(BK.refl) do if p.Parent then pcall(function() p.Reflectance = r end) end end
	for p, s in pairs(BK.sh) do if p.Parent then pcall(function() p.CastShadow = s end) end end
	BK.mat = setmetatable({}, {__mode="k"})
	BK.refl = setmetatable({}, {__mode="k"})
	BK.sh = setmetatable({}, {__mode="k"})
	limparCulling()
end
